511 //dat/dt/a = -K_a*G3_over_c5*m1*m2*(m1+m2)/(a^4 * (1-e^2)^7/2) * (1+c2_a*e^2+c4_a*e^4) (Eq. 5.6 Peter64)
512 const double K_a=64./5., c2_a=73./24., c4_a=37./96.;
513 // det/dt/e = -K_e*G3_over_c5*m1*m2*(m1+m2)/(a^4 * (1-e^2)^5/2) * (1+c2_e*e^2) (Eq. 5.7 Peter64)
514 const double K_e=304./15., c2_e=121./304.;
515
523 inline double function_core(double M1, double M2, double a){
524
525 double a4=a*a*a*a;
526
527 return G3_over_c5*M1*M2*(M1+M2)/a4;
528
529 }

531
540 inline double Peters_GW_de(double M1, double M2, double a, double e){
541
542 double cost=K_e*function_core(M1, M2, a);
543 double e2=e*e;
544
545 double num=(1+c2_e*e2);
546 double den=(1-e2);
547 den=std::sqrt(den*den*den*den*den); //(1-e^2)^(5/2)
548
549 return -cost*num/den;
550
551 }
552
553
562 inline double Peters_GW_da(double M1, double M2, double a, double e){
563
564 double cost=K_a*function_core(M1, M2, a);
565 double e2=e*e;
566 double e4=e2*e2;
567 double den=(1-e2);
568 den=std::sqrt(den*den*den*den*den*den*den); //(1-e^2)^(7/2)
569
570 double num=(1+c2_a*e2+c4_a*e4);
571
572 return -cost*num/den;
573
574 }
