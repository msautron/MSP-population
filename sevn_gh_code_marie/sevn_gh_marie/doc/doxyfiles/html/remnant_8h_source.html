132 inline double XspinGeneva(const double z0, const double mco) const {
133
134 double a, b, m1, m2, alow;
135
136 a = -0.088;
137 if (z0 > 0.01)
138 b = 2.258, m1 = 16.0, alow = 0.13; //m2 = 24.2
139 else if (z0 > 0.004) // 0.004 < z0 <= 0.01
140 b = 3.578, m1 = 31.0, alow = 0.25; //m2 = 37.8
141 else if (z0 > 0.0012) // 0.0012 < z0 <= 0.004
142 b = 2.434, m1 = 18.0, alow = 0.0; //m2 = 27.7
143 else // z0 <= 0.0012
144 b = 3.666, m1 = 32.0, alow = 0.25; //m2 = 38.8
145
146 m2 = (alow - b) / a;
147 if (mco <= m1)
148 return 0.85;
149 else if (mco < m2)
150 return a * mco + b;
151 else
152 return alow;
153 }

164 inline double XspinMESA(const double z0, const double mco) const {
165
166 double a1, b1, a2, b2, m1;
167
168
169 if (z0 > 0.0012 && z0 <= 0.004){
170 a1 = 0.0076, b1 = 0.05, a2 = -0.0019, b2 = 0.165, m1 = 12.09;
171 if (mco <= m1)
172 return a1 * mco + b1;
173 else
174 return a2 * mco + b2;
175 }
176 else if (z0 > 0.01)
177 a1 = -0.0016, b1 = 0.115;
178 else if (z0 > 0.004) // 0.004 < z0 <= 0.01
179 a1 = -0.0006, b1 = 0.105;
180 else // z0 <= 0.0012
181 a1 = -0.0010, b1 = 0.125;
182
183 //Fig. 2 of Belczynski1+20 (https://www.aanda.org/articles/aa/pdf/2020/04/aa36528-19.pdf) shows
184 //that the fitting equations are valid only up to approx a MCO mass of 70 Msun, but Xspin seems
185 //to drop to 0 for large MCO, therefore we just extraploate putting a lower limit at 0 Msun
186
187 return std::max(a1 * mco + b1,0.0);
188 }

205 inline double XspinMaxwellian(const double sigma_xspin) const {
206
207 double x1, x;
208
209 std::normal_distribution<double> gaussian_xspin{0.0, sigma_xspin};
210
211 x1 = gaussian_xspin(utilities::mtrand);
212 x1 *= x1;
213 x = gaussian_xspin(utilities::mtrand);
214 x *= x;
215 x += x1;
216 x1 = gaussian_xspin(utilities::mtrand);
217 x1 *= x1;
218 x += x1;
219
220 x = sqrt(x);
221 if (x > 1.0) /* Just in case the distribution throws x > 1 (possible although not too probable?)*/
222 x = 1.0;
223
224 return x;
225 }
