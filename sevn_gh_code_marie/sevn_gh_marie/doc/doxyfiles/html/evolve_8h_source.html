655 class EvolveBLC : public EvolveFunctor {
656 public:
657 explicit EvolveBLC(SevnLogging& svlog, bool record_state=true, bool include_failed=false)
658 : EvolveFunctor(svlog,record_state,include_failed) {};
659 inline int operator() (Binstar& binary) override {
660
661 bool BLC_flag = false, BLC_case1=false, BLC_case2=false;
662 //The BLC counter counts the evolution steps the systems has been in the BLC configuration
663 //It is increased by one if the BLC_flag is zero, while it is reset to zero when BLC_flag is false
664 unsigned int BLC_counter=0;
665 bool evolution_result = EXIT_SUCCESS; //Reset to true
666
667 for(;;) {
668
669 try {
670
672 binary.evolve();
673
674 BLC_case1 = binary.getstar(0)->amiCompact() and !binary.getstar(1)->amiCompact();
675 BLC_case2 = binary.getstar(1)->amiCompact() and !binary.getstar(0)->amiCompact();
676
678 BLC_flag = (BLC_case1 or BLC_case2) and !binary.broken;
679
680 //If BLC_flag is true, update the BLC counter and record state (if allowed by other flags)
681 if (BLC_flag){
682 BLC_counter++;
683 //Record only if it is time (or print only end and this is the start of the BLC conditions BLC_counter=) and if _record state is true
684 if ( ( (binary.isoutputtime() or binary.printall()) or BLC_counter==1 ) and _record_state ) {
685 binary.recordstate_w_timeupdate();
686 }
687 }
688 //If BLC_flag is false, but BLC_counter>0 we are at the end of a BLC phase, so record state (if allowed by other flags) ans reset the BLC counter
689 //FInally break, in fact if this system was in a BLC phase and not it is not, there is no way it will enter again in a BLC phase
690 else if(BLC_counter>0){
691
692 //Always print the end
693 if (_record_state){
694 binary.recordstate_w_timeupdate();
695 }
696 //reset the BLC counter
697 BLC_counter=0;
698 break;
699 }
700
701 if (binary.breaktrigger()) //stopping condition for evolving the star
702 break; //go to evolve the next star
703
704 }
705 catch(sevnstd::sevnerr& e){
706
707 catched_error_message(binary,e);
708
709 evolution_result=EXIT_FAILURE;
710 break;
711 }
712 }
713 /********************************/
714 final_print(binary,evolution_result); //print out all the recorded states for the star
715 return evolution_result;
716 }

719 sevnstd::notimplemented_error());
720 return -1;
721 }

978 template <typename System>
979 inline int evolve_list(EvolveFunctor* evolve_function, std::vector<System>& systems, _UNUSED IO& sevnio, int Nevolve=-1){
980 SevnLogging svlog;
981
982 if (Nevolve==-1) Nevolve=systems.size();
983 unsigned Nfailed =0;
984
985#ifdef _OPENMP
986#pragma omp parallel num_threads(sevnio.nthreads)
987#endif
988 {
989
990#ifdef _OPENMP
991#pragma omp for schedule(static) reduction(+: Nfailed)
992#endif
993 for (size_t i = 0; i < (size_t)Nevolve; i++) {
994
995 //NOTICE: THE FOLLOWING IS EXTREMELY IMPORTANT
996 //mtrand is a static threadlocal variaible, i.e. it is a global variable for each thread
997 //it is "seeded" each time we initiliase a system (binary or star), but we initiliase them
998 //before to evolve them, therefore is you not "re-seed" mtrand before the evolution
999 //mtrand will be seeded with the last loaded system and this will make impossibile to reproduce our results
1000 //unless we use exactly the same list of systems (at the same relative position in the input file)
1001 //Notice also that it is important that mtrand IS NEVER RE-SEEDED during the stellar/binary evolution
1002 //(for this reason when we create an auxiliary star we save the current mtrand status, then we initiliase the
1003 //auxiliary star where mtrand is reseed and finally we restore the old mtrand)
1004 //TODO We should surely improve this behaviour (see issue 71 on gitlab: https://gitlab.com/sevn/SEVN/-/issues/71 )
1005 utilities::mtrand.seed(systems[i].get_rseed());//Set random state for riproducibility, DO NOT MOVE OR REMOVE THIS LINE
1006
1007
1008 /****** EVOLVE ******/
1009 if((*evolve_function)(systems[i])==EXIT_FAILURE)
1010 Nfailed++;
1011
1012
1013 }
1014
1015
1016 }
