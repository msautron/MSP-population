399 inline double hurley_rm_x(double Z){
400
401 double zeta = std::log(Z/0.02);
402
403 double ret = 0.30406 + 0.0805*zeta + 0.0897*zeta*zeta + 0.0878*zeta*zeta*zeta + 0.0222*zeta*zeta*zeta*zeta;
404
405
406 return ret;
407
408 }

410 double hurley_final_mass(const double Ebindf, const double toll=1e-3, const double maxit=100){
411
412 //Check
413 if (!isinisialised)
414 svlog.critical("CE process has not been initialised",__FILE__,__LINE__,sevnstd::ce_error());
415
416 double XX = 1 + hurley_rm_x(primary->get_Z());
417
418
419 double MC22=secondary->getp(MHE::ID);
420 double MC1=M_core_primary;
421 double M1=primary->getp(Mass::ID);
422 double M2=secondary->getp(Mass::ID);
423 double DELY, DERI;
424 int Niter = 0;
425
426 double Eratio = Ebindf/Ebind_ini;
427
428 double CONST = (pow(M1+M2,XX))*(M1-MC1+M2-MC22)*Eratio;
429
430 double MF=std::max(MC1+MC22, (M1+M2)*pow((Eratio),(1/XX)));
431 DELY = 10*toll*MF;
432
433
434 while(std::abs(DELY/MF)>toll && Niter<maxit){
435
436 DELY = pow(MF,XX)*(MF-MC1-MC22) - CONST;
437
438 DERI = pow(MF,(XX-1))*((1+XX)*MF - XX*(MC1+MC22));
439
440 MF -= DELY/DERI;
441
442 Niter++;
443
444
445
446 }
447
448
449
450 return MF;
451
452 }
