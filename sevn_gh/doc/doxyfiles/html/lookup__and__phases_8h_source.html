208 /***************************************************
209 * GI 230120: Map to handle the input options
210 * This part handle the input params for the binary evolution.
211 * All the possibile modality are reported in the enum InputBinaryOption.
212 * I attached to the modalities the expected number of parameters to read from the file
213 * to be able to properly handle the possible errors and exceptions.
214 * This info is stored in the map INPUTMAPBIN_PARAM.
215 * Finally the map INPUTMAPBIN maps a string that can be given as option in io.load to
216 * a pair containing the Input modality and the associated number of expected parameters.
217 **************************************************/
218 //GI 230120: Map to manage the input options
219 typedef std::pair<InputBinaryOption,int> _INPAIR; //PAIR INPUT OPTION - NUMBER OF EXPECTED INPUT PARAMETERS
220 typedef std::map<InputBinaryOption, int> INPUTMAPBIN_NPARAM; //Map between Input option and Number of expected input parameters
221 typedef std::map<std::string, _INPAIR> INPUTMAPBIN; //MAP string - PAIR (Input Option, N expected)
222 //They are defined in cpp
223 extern const INPUTMAPBIN inputmapbin;
224 extern const INPUTMAPBIN_NPARAM inputmapbin_nparam;
225 /***************************************************/
